```python
"""FR6.2 P1：规则簇 → LLM 精分类（逐簇细化、逐簇降级，source 可审计）."""
from __future__ import annotations

import json
import logging
from typing import TYPE_CHECKING, Literal

from pydantic import BaseModel

from rsi.core.models import AttributionReport

if TYPE_CHECKING:
    from rsi.llm.client import LLMClient
    from rsi.optimizer.attribution import AttributionAnalyzer

logger = logging.getLogger(__name__)

_RUN_SUMMARY_CAP = 800


class RefinedClusterOut(BaseModel):
    failure_mode: str
    defect_class: Literal["A", "B"]
    evidence_refs: list[str] = []
    rationale: str = ""
    directions: list[str] = []


class LLMRefinedAttributionAnalyzer:
    def __init__(self, base, llm, *, max_clusters: int = 5):
        self._base = base
        self._llm = llm
        self._max_clusters = max_clusters

    async def analyze(self, failed_runs, sut_config, opt_history):
        report = await self._base.analyze(failed_runs, sut_config, opt_history)
        refined_any = False
        directions: list[str] = []
        by_id = {r.get("run_id"): r for r in (failed_runs or [])}
        for cluster in report.clusters[: self._max_clusters]:
            samples = []
            for rid in cluster.sample_run_ids[:5]:
                run = by_id.get(rid, {})
                cmds = [
                    e.get("payload", {}).get("command", "")
                    for e in run.get("events", [])
                    if e.get("kind") == "TOOL_CALL"
                ]
                samples.append(f"{rid}: {'; '.join(cmds)}"[:_RUN_SUMMARY_CAP])
            if not samples:
                continue
            sut_ev = "\n".join(
                f"{k}: {v[:400]}" for k, v in (sut_config or {}).items()
            )[:4000]
            vars = {
                "cluster_samples": "\n".join(samples),
                "failure_modes": cluster.failure_mode,
                "opt_history": json.dumps(opt_history or [], ensure_ascii=False),
                "sut_evidence": sut_ev or "（无配置原文）",
            }
            try:
                resp = await self._llm.complete(
                    "attribution", "classify", vars, schema=RefinedClusterOut
                )
            except Exception as exc:  # noqa: BLE001 - 逐簇降级
                logger.warning("attribution.llm_refine_failed",
                               mode=cluster.failure_mode, error=str(exc))
                continue
            out: RefinedClusterOut = resp.parsed
            cluster.failure_mode = out.failure_mode or cluster.failure_mode
            cluster.defect_class = out.defect_class
            cluster.evidence = list(dict.fromkeys(
                list(cluster.evidence) + list(out.evidence_refs)))
            cluster.rationale = out.rationale or None
            directions.extend(d for d in out.directions if d)
            refined_any = True
        if refined_any:
            report.source = "llm"
            report.directions = list(dict.fromkeys(directions))
        return report
```
