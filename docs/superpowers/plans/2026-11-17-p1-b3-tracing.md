`platform/rsi/tracing/redact.py`：

```python
"""FR2.7 脱敏管道（DD-07 §4 Redactor / §5.3 管道位：摄取后、seq 分配前）.

wiretap 原文永不改动（审计原文）；脱敏只作用于解析后的 TraceEvent——
events.ndjson、trace_events_index、WS 推流看到的都是脱敏后内容。
指纹精确匹配（secret 原值 + base64 → ***，仅命中子串替换）+ 规则正则
（内建通用模式 + rules_path yaml 自定义规则）。
"""

from __future__ import annotations

import re
from pathlib import Path
from typing import Any, Callable, Iterable

import yaml

from rsi.core.models import TraceEvent
from rsi.log import get_logger
from rsi.security.credentials import SecretFingerprint

__all__ = ["MASK", "Redactor"]

logger = get_logger(__name__)

MASK = "***"
# 短串精确替换会大面积误伤（离线夹具存在 password="pw"）——只替换 >=6 字符明文
_MIN_EXACT_LEN = 6

# 内建通用规则：赋值形态只掩值部；常见 key 前缀整体掩码（保守集）
_ASSIGN_RE = re.compile(
    r"(?i)\b(password|passwd|pwd|secret|token|api[_-]?key)\b(\s*[=:]\s*)(\S+)"
)
_BUILTIN_RULES: list[tuple[str, re.Pattern[str]]] = [
    ("bearer_token", re.compile(r"Bearer\s+[A-Za-z0-9._\-]{8,}")),
    ("openai_style_key", re.compile(r"sk-[A-Za-z0-9\-_]{16,}")),
]


class Redactor:
    def __init__(
        self,
        fingerprint_provider: Callable[[], Iterable[SecretFingerprint]] | None = None,
        rules_path: Path | None = None,
    ) -> None:
        self._provider = fingerprint_provider
        self._rules = _load_rules(rules_path)

    def redact(
        self,
        ev: TraceEvent,
        fingerprints: Iterable[SecretFingerprint] = (),
    ) -> TraceEvent:
        """DD-07 §4 签名：run 级指纹 ∪ 静态指纹（provider）精确替换 + 规则正则。"""
        fps = list(fingerprints)
        if self._provider is not None:
            fps += list(self._provider())
        data = _scrub(ev.data, fps, self._rules)
        if data == ev.data:
            return ev
        return ev.model_copy(update={"data": data})


def _load_rules(rules_path: Path | None) -> list[tuple[str, re.Pattern[str]]]:
    rules = list(_BUILTIN_RULES)
    if rules_path is None:
        return rules
    path = Path(rules_path)
    if not path.is_file():
        return rules
    try:
        doc = yaml.safe_load(path.read_text(encoding="utf-8")) or {}
    except Exception as exc:  # noqa: BLE001 - 规则文件坏不阻断摄取
        logger.warning("redact.rules_load_failed", path=str(path), error=str(exc))
        return rules
    for item in doc.get("rules") or []:
        if isinstance(item, dict) and item.get("pattern"):
            name = str(item.get("name") or "custom")
            try:
                rules.append((name, re.compile(str(item["pattern"]))))
            except re.error as exc:
                logger.warning("redact.rule_invalid", name=name, error=str(exc))
    return rules


def _scrub(value: Any, fps: list[SecretFingerprint], rules) -> Any:
    if isinstance(value, dict):
        return {k: _scrub(v, fps, rules) for k, v in value.items()}
    if isinstance(value, (list, tuple)):
        return [_scrub(v, fps, rules) for v in value]
    if isinstance(value, str):
        return _mask_str(value, fps, rules)
    return value


def _mask_str(s: str, fps: list[SecretFingerprint], rules) -> str:
    for fp in fps:
        for p in fp.patterns:
            if p and len(p) >= _MIN_EXACT_LEN:
                s = s.replace(p, MASK)
    s = _ASSIGN_RE.sub(r"\1\2" + MASK, s)
    for _name, rx in rules:
        s = rx.sub(MASK, s)
    return s
```

```python
    def _runs_by_join(self, f: TraceFilter) -> set[str] | None:
        """batch/config_version → run_id 集（runs⋈batches）；无联结维度返回 None。"""
        if not (f.batch_id or f.config_version_id):
            return None
        if self._session_factory is None:
            logger.warning(
                "trace.join_no_db",
                batch_id=f.batch_id, config_version_id=f.config_version_id,
            )
            return set()
        try:
            with self._session_factory() as session:
                stmt = select(Run.id).join(Batch, Run.batch_id == Batch.id)
                if f.batch_id:
                    stmt = stmt.where(Run.batch_id == f.batch_id)
                if f.config_version_id:
                    stmt = stmt.where(Batch.config_version_id == f.config_version_id)
                return set(session.exec(stmt).all())
        except Exception as exc:  # noqa: BLE001 - 查询降级空集
            logger.warning("trace.join_failed", error=str(exc))
            return set()

    def _universe(self) -> set[str]:
        """run 全集 = 内存 buffer ∪ 落盘目录 ∪ DB 索引（重启后兜底）。"""
        runs = set(self._collector.run_ids())
        base = self._collector.persistence_dir
        if base is not None and Path(base).is_dir():
            runs |= {
                p.name for p in Path(base).iterdir()
                if p.is_dir() and (p / "events.ndjson").is_file()
            }
        if self._session_factory is not None:
            try:
                with self._session_factory() as session:
                    runs |= set(
                        session.exec(select(TraceEventIndex.run_id).distinct()).all()
                    )
            except Exception as exc:  # noqa: BLE001
                logger.debug("trace.universe_db_failed", error=str(exc))
        return runs

    def _events_of_run(self, run_id: str) -> list[TraceEvent]:
        """内容水合：内存 buffer → events.ndjson 冷读（run 级缓存，上限 64 run）。"""
        mem = self._collector.events(run_id)
        if mem:
            return mem
        if run_id in self._file_cache:
            return self._file_cache[run_id]
        base = self._collector.persistence_dir
        if base is None:
            return []
        path = Path(base) / run_id / "events.ndjson"
        if not path.is_file():
            return []
        parser = NdjsonEventParser()
        out: list[TraceEvent] = []
        for line in path.read_text(encoding="utf-8").splitlines():
            if line.strip():
                e = parser.parse_line(line, defaults={"run_id": run_id})
                if e is not None:
                    out.append(e)
        if len(self._file_cache) >= 64:
            self._file_cache.clear()
        self._file_cache[run_id] = out
        return out
```

```python
    body: TraceQueryBody,
    request: Request,
    user: Claims = Depends(get_current_user),
    traces: dict = Depends(get_store("traces")),
) -> TraceQueryOut:

    # 统一后过滤（内存注入行是 dict；store 行已过滤过一遍，重复过滤幂等）
    if body.event_types is not None or body.type is not None:
        wanted = set(body.event_types or [body.type])
        events = [e for e in events if e.get("type") in wanted]
    if body.is_error is not None:
        events = [e for e in events if (e.get("type") == "error") == body.is_error]
    for key in ("task_id", "lab_id", "sut_id"):
        val = getattr(body, key)
        if val is not None:
            events = [e for e in events if e.get(key) == val]
    if body.from_ts is not None:
        events = [e for e in events if e.get("ts", "") >= body.from_ts]
    if body.to_ts is not None:
        events = [e for e in events if e.get("ts", "") <= body.to_ts]
    if body.since_seq is not None:
        events = [e for e in events if e.get("seq", 0) >= body.since_seq]
    if body.cursor is not None:
        cur_run, _, cur_seq = body.cursor.partition(":")
        events = [
            e for e in events
            if (e.get("run_id", ""), e.get("seq", 0)) > (cur_run, int(cur_seq))
        ]
    limit = min(body.limit, 10_000)
    page = events[:limit]
    next_cursor = None
    if len(events) > limit and page:
        last = page[-1]
        next_cursor = f"{last.get('run_id', '')}:{last.get('seq', 0)}"
    return TraceQueryOut(items=page, next_cursor=next_cursor)
```

```python
def _merged_events(request: Request, traces: dict, f: "TraceFilter") -> list[dict[str, Any]]:
    """内存 traces store（测试注入）优先，TraceStore（collector/DB/文件）补缺
    ——同 run 以内存为准（W12 兜底结构保留）。"""
    out: list[dict[str, Any]] = []
    mem_runs: dict[str, list] = (
        {rid: traces[rid] for rid in f.run_ids if rid in traces}
        if f.run_ids else dict(traces)
    )
    for rid, rows in mem_runs.items():
        out.extend({**e, "run_id": rid} for e in rows)
    mem_have = set(mem_runs)
    for e in _store_events(request, f):
        if e.get("run_id") not in mem_have:
            out.append(e)
    return sorted(out, key=lambda e: (e.get("run_id", ""), e.get("seq", 0)))
```

```python
_SIG_DIGITS = re.compile(r"\d+")
_SIG_HEX = re.compile(r"\b[0-9a-f]{6,}\b", re.IGNORECASE)


def _error_signature(ev: TraceEvent) -> str:
    """错误消息归一化签名：数字→#、十六进制串→<hex>、压空白、小写、截 120。"""
    d = ev.data or {}
    raw = d.get("msg") or d.get("error") or d.get("message") or ""
    if not isinstance(raw, str):
        raw = json.dumps(raw, ensure_ascii=False, default=str)
    s = _SIG_HEX.sub("<hex>", _SIG_DIGITS.sub("#", raw))
    s = re.sub(r"\s+", " ", s).strip().lower()
    return s[:120] or "<empty>"


def _tool_of(ev: TraceEvent) -> str:
    d = ev.data or {}
    return str(d.get("tool") or d.get("tool_name") or d.get("name") or "<unknown>")


def _event_sample(ev: TraceEvent) -> str:
    d = ev.data or {}
    msg = d.get("msg") or d.get("error") or d.get("message")
    if isinstance(msg, str) and msg:
        return msg[:200]
    return f"{ev.event_type.value} seq={ev.seq} ts={ev.ts}"
```

```python
    def cluster_failures(
        self,
        f: TraceFilter | None = None,
        group_by: str = "error_signature",
        sample_limit: int = 5,
    ) -> list[dict[str, Any]]:
        """FR2.5/FR10.9⑥：失败轨迹规则聚类（DD-07:67 签名），供 FR6.2 消费。"""
        if group_by not in ("error_signature", "tool_name", "task_id"):
            raise ValidationError(
                f"unknown group_by {group_by!r}",
                context={"group_by": group_by,
                         "allowed": ["error_signature", "tool_name", "task_id"]},
            )
        errs = [
            e for e in self._filter_events(f or TraceFilter())
            if e.event_type == EventType.ERROR
        ]
        groups: dict[str, list[TraceEvent]] = {}
        for e in errs:
            key = (
                _error_signature(e) if group_by == "error_signature"
                else _tool_of(e) if group_by == "tool_name"
                else e.task_id
            )
            groups.setdefault(key, []).append(e)
        out: list[dict[str, Any]] = []
        for feature, evs in groups.items():
            out.append({
                "feature": feature,
                "group_by": group_by,
                "run_ids": sorted({e.run_id for e in evs}),
                "event_count": len(evs),
                "samples": [_event_sample(e) for e in evs[:sample_limit]],
            })
        out.sort(key=lambda c: (-c["event_count"], c["feature"]))
        return out
```

```python
"""FR2.6 存储管理：zstd 归档 + 老化（DD-07 §7-4）.

hot: <traces_dir>/<run_id>/ 目录（在线可查，不动）；
archive: 已 finalize（meta.json 存在）且超 tracing.archive.hot_days →
         <run_id>.tar.zst（codec 可配 zstd|gzip|none），原目录随后移除；
cold: 归档文件 mtime 超 tracing.archive.cold_days（>0 才启用）删除；
promoted: experiments.decision=promoted 引用批次的 run 按
          keep_promoted_forever 跳过（永久保留证据链）。
"""

from __future__ import annotations

import json
import shutil
import tarfile
from datetime import datetime, timedelta, timezone
from pathlib import Path
from typing import Any, Callable

from rsi.log import get_logger

if True:  # sqlmodel 延迟到需要 DB 的路径再 import 顶部常驻（doctor 独立复用 scan 不拉 DB）
    pass

__all__ = ["TraceArchiver", "scan_storage"]

logger = get_logger(__name__)

_SUFFIX = {"zstd": ".tar.zst", "gzip": ".tar.gz", "none": ".tar"}


def scan_storage(traces_dir: Path) -> dict[str, Any]:
    """存储量统计（doctor / system 报告用；无 DB 依赖）。"""
    d = Path(traces_dir)
    hot_runs = archived_files = 0
    hot_bytes = archived_bytes = 0
    if d.is_dir():
        for p in d.iterdir():
            if p.is_dir():
                hot_runs += 1
                hot_bytes += sum(
                    f.stat().st_size for f in p.rglob("*") if f.is_file()
                )
            elif p.is_file():
                archived_files += 1
                archived_bytes += p.stat().st_size
    return {
        "hot_runs": hot_runs,
        "hot_bytes": hot_bytes,
        "archived_files": archived_files,
        "archived_bytes": archived_bytes,
    }


class TraceArchiver:
    def __init__(
        self,
        traces_dir: Path,
        archive_settings: Any,
        session_factory: Any | None = None,
        now: Callable[[], datetime] | None = None,
    ) -> None:
        self._dir = Path(traces_dir)
        self._cfg = archive_settings
        self._session_factory = session_factory
        self._now = now or (lambda: datetime.now(timezone.utc))

    # -- run 保护名单 ------------------------------------------------------

    def _protected_run_ids(self) -> set[str]:
        if not getattr(self._cfg, "keep_promoted_forever", True):
            return set()
        if self._session_factory is None:
            return set()
        from sqlmodel import select

        from rsi.db.tables import Experiment, Run

        try:
            with self._session_factory() as session:
                rows = session.exec(
                    select(Experiment).where(Experiment.decision == "promoted")
                ).all()
                batch_ids = {
                    bid
                    for r in rows
                    for bid in (r.candidate_batch_id, r.baseline_batch_id)
                    if bid
                }
                if not batch_ids:
                    return set()
                return set(
                    session.exec(
                        select(Run.id).where(Run.batch_id.in_(batch_ids))  # type: ignore[attr-defined]
                    ).all()
                )
        except Exception as exc:  # noqa: BLE001 - 名单查不出 → 照常归档（压缩无损）
            logger.warning("archive.promoted_query_failed", error=str(exc))
            return set()

    # -- 主流程 ------------------------------------------------------------

    def archive_pass(self) -> dict[str, Any]:
        result: dict[str, Any] = {
            "archived": [], "deleted": [],
            "skipped_promoted": 0, "errors": 0,
        }
        if not self._dir.is_dir():
            return result
        now = self._now()
        hot_cut = now - timedelta(days=self._cfg.hot_days)
        protected = self._protected_run_ids()
        for run_dir in sorted(p for p in self._dir.iterdir() if p.is_dir()):
            run_id = run_dir.name
            meta_p = run_dir / "meta.json"
            if not meta_p.is_file():
                continue  # 未 finalize 的在线 run——DD-07 §7-4：在线数据不动
            age_ref = _meta_last_ts(meta_p) or datetime.fromtimestamp(
                run_dir.stat().st_mtime, tz=timezone.utc
            )
            if age_ref >= hot_cut:
                continue
            if run_id in protected:
                result["skipped_promoted"] += 1
                continue
            dest = run_dir.with_name(run_id + _SUFFIX[self._cfg.codec])
            try:
                _pack(run_dir, dest, self._cfg.codec)
                self._update_trace_files(run_id, dest)
                shutil.rmtree(run_dir)
            except Exception as exc:  # noqa: BLE001 - 单 run 失败不阻断整轮
                result["errors"] += 1
                logger.warning(
                    "archive.pack_failed", run_id=run_id, error=str(exc)
                )
                continue
            result["archived"].append(str(dest))
            logger.info("archive.packed", run_id=run_id, dest=str(dest))
        if self._cfg.cold_days > 0:
            cold_cut = now - timedelta(days=self._cfg.cold_days)
            for f in sorted(self._dir.glob("*.tar.*")):
                if datetime.fromtimestamp(
                    f.stat().st_mtime, tz=timezone.utc
                ) >= cold_cut:
                    continue
                try:
                    f.unlink()
                except OSError as exc:
                    result["errors"] += 1
                    logger.warning("archive.prune_failed", file=str(f), error=str(exc))
                    continue
                result["deleted"].append(str(f))
                logger.info("archive.pruned", file=str(f))
        return result

    def _update_trace_files(self, run_id: str, dest: Path) -> None:
        """归档后 ndjson_path 指到归档文件（DB 失败不阻断——目录已删，登记尽力）。"""
        if self._session_factory is None:
            return
        from rsi.db.tables import TraceFile

        try:
            with self._session_factory() as session:
                row = session.get(TraceFile, run_id)
                if row is not None:
                    row.ndjson_path = str(dest)
                    row.bytes = dest.stat().st_size
                    session.add(row)
                    session.commit()
        except Exception as exc:  # noqa: BLE001
            logger.warning(
                "archive.trace_files_update_failed", run_id=run_id, error=str(exc)
            )


def _meta_last_ts(meta_p: Path) -> datetime | None:
    try:
        doc = json.loads(meta_p.read_text(encoding="utf-8"))
        ts = doc.get("last_ts")
        return datetime.fromisoformat(ts) if ts else None
    except Exception:  # noqa: BLE001 - meta 坏走目录 mtime 兜底
        return None


def _pack(run_dir: Path, dest: Path, codec: str) -> None:
    files = sorted(p for p in run_dir.rglob("*") if p.is_file())
    if codec == "zstd":
        import zstandard

        cctx = zstandard.ZstdCompressor()
        with open(dest, "wb") as raw:
            with cctx.stream_writer(raw) as compressed:
                with tarfile.open(fileobj=compressed, mode="w|") as tf:
                    for p in files:
                        tf.add(p, arcname=p.relative_to(run_dir).as_posix())
    else:
        mode = "w:gz" if codec == "gzip" else "w"
        with tarfile.open(dest, mode) as tf:
            for p in files:
                tf.add(p, arcname=p.relative_to(run_dir).as_posix())
```

```python
_ELEMENT_LINE_CAP = 1500
_ELEMENT_TOTAL_CAP = 8000
_ELEMENT_PATH_CAP = 200


def _render_element_texts(texts: dict[str, str]) -> str:
    """dict[str, str] → ``"path[:200]: content[:1500]"`` 多行文本，总长 cap 8000。

    按 path 排序保证确定性（cassette 键与测试断言依赖稳定顺序）；
    超出总长时尾部条目省略，但追加显式省略标记行（B2 终审遗留：
    静默丢尾会让 LLM 误以为清单完整）。
    """
    lines: list[str] = []
    total = 0
    omitted = 0
    for path in sorted(texts):
        line = (
            f"{path[:_ELEMENT_PATH_CAP]}: "
            f"{(texts[path] or '')[:_ELEMENT_LINE_CAP]}"
        )
        if total + len(line) > _ELEMENT_TOTAL_CAP:
            omitted = len(texts) - len(lines)
            break
        lines.append(line)
        total += len(line)
    if omitted:
        lines.append(f"...（其余 {omitted} 个条目因长度上限省略）")
    return "\n".join(lines)
```
